DROP DATABASE IF EXISTS caishen_food_lab;
CREATE DATABASE caishen_food_lab DEFAULT CHARACTER SET utf8mb4 COLLATE utf8mb4_general_ci;
USE caishen_food_lab;

-- =====================================================================
-- 一、用户与偏好
-- =====================================================================
CREATE TABLE t_user (
  id            BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  account       VARCHAR(64)  NOT NULL COMMENT '登录账号',
  password      VARCHAR(128) NOT NULL COMMENT '密码(BCrypt)',
  nickname      VARCHAR(64)      NULL COMMENT '昵称',
  phone         VARCHAR(20)      NULL COMMENT '手机号',
  avatar        VARCHAR(255)     NULL COMMENT '头像',
  status        TINYINT      NOT NULL DEFAULT 1 COMMENT '1正常 0停用',
  create_time   DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP,
  update_time   DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  UNIQUE KEY uk_user_account (account)
) ENGINE=InnoDB COMMENT='用户';

CREATE TABLE t_user_preference (
  id               BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  user_id          BIGINT UNSIGNED NOT NULL COMMENT '用户ID',
  tastes           VARCHAR(128) NOT NULL DEFAULT '' COMMENT '口味偏好(多选,逗号分隔:酸,甜,辣,咸,清淡)',
  diet_type        VARCHAR(16)  NOT NULL DEFAULT '不限' COMMENT '饮食方式:不限/素食/生酮/低卡',
  default_servings INT          NOT NULL DEFAULT 2 COMMENT '默认用餐人数',
  update_time      DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  UNIQUE KEY uk_pref_user (user_id)
) ENGINE=InnoDB COMMENT='用户饮食偏好';

CREATE TABLE t_user_avoid_item (
  id          BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  user_id     BIGINT UNSIGNED NOT NULL,
  type        TINYINT NOT NULL COMMENT '1忌口食材 2过敏源',
  name        VARCHAR(64) NOT NULL,
  create_time DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  KEY idx_avoid_user (user_id)
) ENGINE=InnoDB COMMENT='用户忌口/过敏源';

CREATE TABLE t_user_ingredient (
  id            BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  user_id       BIGINT UNSIGNED NOT NULL,
  ingredient_id BIGINT UNSIGNED     NULL COMMENT '关联食材库',
  name          VARCHAR(64) NOT NULL,
  quantity      VARCHAR(32)     NULL,
  unit          VARCHAR(16)     NULL,
  create_time   DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  KEY idx_ui_user (user_id)
) ENGINE=InnoDB COMMENT='我的食材篮';

CREATE TABLE t_admin_user (
  id              BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  username        VARCHAR(64)  NOT NULL,
  password        VARCHAR(128) NOT NULL COMMENT '密码(BCrypt)',
  real_name       VARCHAR(64)      NULL,
  status          TINYINT      NOT NULL DEFAULT 1,
  last_login_time DATETIME         NULL,
  create_time     DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  UNIQUE KEY uk_admin_username (username)
) ENGINE=InnoDB COMMENT='后台管理员';

-- =====================================================================
-- 二、食材与营养
-- =====================================================================
CREATE TABLE t_ingredient (
  id           BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  name         VARCHAR(64) NOT NULL COMMENT '食材名',
  alias        VARCHAR(128)    NULL COMMENT '别名(逗号分隔)',
  category     VARCHAR(32) NOT NULL DEFAULT '其他' COMMENT '分类:蔬菜/肉类/蛋类/调味/主食/其他',
  default_unit VARCHAR(16) NOT NULL DEFAULT '克' COMMENT '默认单位',
  image        VARCHAR(255)    NULL,
  status       TINYINT     NOT NULL DEFAULT 1,
  create_time  DATETIME    NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  KEY idx_ing_name (name)
) ENGINE=InnoDB COMMENT='食材库';

CREATE TABLE t_allergen_tag (
  id      BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  name    VARCHAR(64) NOT NULL,
  remark  VARCHAR(255)    NULL,
  status  TINYINT NOT NULL DEFAULT 1,
  PRIMARY KEY (id)
) ENGINE=InnoDB COMMENT='过敏源标签';

CREATE TABLE t_ingredient_allergen (
  id              BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  ingredient_id   BIGINT UNSIGNED NOT NULL,
  allergen_tag_id BIGINT UNSIGNED NOT NULL,
  PRIMARY KEY (id),
  UNIQUE KEY uk_ing_allergen (ingredient_id, allergen_tag_id)
) ENGINE=InnoDB COMMENT='食材-过敏源映射';

CREATE TABLE t_ingredient_nutrition (
  id            BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  ingredient_id BIGINT UNSIGNED NOT NULL,
  calories      DECIMAL(8,2) NOT NULL DEFAULT 0 COMMENT '热量 kcal',
  protein       DECIMAL(8,2) NOT NULL DEFAULT 0 COMMENT '蛋白质 g',
  fat           DECIMAL(8,2) NOT NULL DEFAULT 0 COMMENT '脂肪 g',
  carbs         DECIMAL(8,2) NOT NULL DEFAULT 0 COMMENT '碳水 g',
  unit_desc     VARCHAR(32) NOT NULL DEFAULT '每100g',
  update_time   DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  UNIQUE KEY uk_nutri_ing (ingredient_id)
) ENGINE=InnoDB COMMENT='营养数据';

-- =====================================================================
-- 三、菜谱
-- =====================================================================
CREATE TABLE t_recipe (
  id          BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  name        VARCHAR(64) NOT NULL,
  cover_image VARCHAR(255)    NULL,
  cuisine     VARCHAR(16) NOT NULL DEFAULT '家常菜' COMMENT '菜系:家常菜/创意菜/轻食',
  difficulty  VARCHAR(16) NOT NULL DEFAULT '简单' COMMENT '难度:简单/中等/较难',
  taste       VARCHAR(32) NOT NULL DEFAULT '清淡' COMMENT '口味',
  cook_time   INT         NOT NULL DEFAULT 15 COMMENT '耗时(分钟)',
  description VARCHAR(255)    NULL,
  servings    INT         NOT NULL DEFAULT 2 COMMENT '份量(人份)',
  source      VARCHAR(16) NOT NULL DEFAULT '人工' COMMENT '来源:人工/AI生成',
  status      VARCHAR(16) NOT NULL DEFAULT '已上线' COMMENT '状态:已上线/待审核/已下架',
  calories    DECIMAL(8,2) NOT NULL DEFAULT 0 COMMENT '每份热量 kcal',
  protein     DECIMAL(8,2) NOT NULL DEFAULT 0,
  fat         DECIMAL(8,2) NOT NULL DEFAULT 0,
  carbs       DECIMAL(8,2) NOT NULL DEFAULT 0,
  create_time DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  KEY idx_recipe_status (status)
) ENGINE=InnoDB COMMENT='菜谱';

CREATE TABLE t_recipe_ingredient (
  id           BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  recipe_id    BIGINT UNSIGNED NOT NULL,
  ingredient_id BIGINT UNSIGNED    NULL,
  name         VARCHAR(64) NOT NULL,
  amount       VARCHAR(32)     NULL COMMENT '数量(如 2 / 少许)',
  unit         VARCHAR(16)     NULL COMMENT '单位',
  is_main      TINYINT NOT NULL DEFAULT 0 COMMENT '1主料',
  is_required  TINYINT NOT NULL DEFAULT 1 COMMENT '1必备 0可替代',
  sort         INT     NOT NULL DEFAULT 0,
  PRIMARY KEY (id),
  KEY idx_ri_recipe (recipe_id)
) ENGINE=InnoDB COMMENT='菜谱所需食材';

CREATE TABLE t_recipe_step (
  id          BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  recipe_id   BIGINT UNSIGNED NOT NULL,
  step_no     INT NOT NULL,
  title       VARCHAR(64)  NOT NULL,
  description VARCHAR(500)     NULL,
  image_url   VARCHAR(255)     NULL COMMENT '步骤示意图',
  tips        VARCHAR(255)     NULL COMMENT '关键提示',
  sort        INT NOT NULL DEFAULT 0,
  PRIMARY KEY (id),
  KEY idx_step_recipe (recipe_id)
) ENGINE=InnoDB COMMENT='分步教学(图文)';

CREATE TABLE t_ingredient_substitute (
  id                  BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  recipe_ingredient_id BIGINT UNSIGNED NOT NULL,
  ingredient_id       BIGINT UNSIGNED    NULL,
  name                VARCHAR(64) NOT NULL,
  ratio_desc          VARCHAR(128)    NULL COMMENT '替代比例',
  note                VARCHAR(255)    NULL COMMENT '口味衔接',
  PRIMARY KEY (id),
  KEY idx_sub_ri (recipe_ingredient_id)
) ENGINE=InnoDB COMMENT='食材替代';

CREATE TABLE t_recipe_favorite (
  id          BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  user_id     BIGINT UNSIGNED NOT NULL,
  recipe_id   BIGINT UNSIGNED NOT NULL,
  create_time DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  UNIQUE KEY uk_fav (user_id, recipe_id)
) ENGINE=InnoDB COMMENT='菜谱收藏';

CREATE TABLE t_recipe_review (
  id          BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  user_id     BIGINT UNSIGNED NOT NULL,
  recipe_id   BIGINT UNSIGNED NOT NULL,
  rating      TINYINT NOT NULL DEFAULT 5 COMMENT '1-5星',
  tags        VARCHAR(128) NULL COMMENT '评价标签(选填)',
  content     VARCHAR(500) NULL,
  status      VARCHAR(16) NOT NULL DEFAULT '待审核' COMMENT '待审核/已通过/已驳回',
  create_time DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  KEY idx_review_status (status)
) ENGINE=InnoDB COMMENT='菜谱评价';

CREATE TABLE t_recipe_history (
  id        BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  user_id   BIGINT UNSIGNED NOT NULL,
  recipe_id BIGINT UNSIGNED NOT NULL,
  view_time DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  KEY idx_hist_user (user_id)
) ENGINE=InnoDB COMMENT='浏览历史';

-- =====================================================================
-- 四、购物清单与订单
-- =====================================================================
CREATE TABLE t_shopping_list (
  id             BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  user_id        BIGINT UNSIGNED NOT NULL,
  name           VARCHAR(64) NOT NULL DEFAULT '我的购物清单',
  source_recipes VARCHAR(255) NULL COMMENT '来源菜谱(逗号分隔)',
  create_time    DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  KEY idx_sl_user (user_id)
) ENGINE=InnoDB COMMENT='购物清单';

CREATE TABLE t_shopping_list_item (
  id            BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  list_id       BIGINT UNSIGNED NOT NULL,
  ingredient_id BIGINT UNSIGNED    NULL,
  name          VARCHAR(64) NOT NULL,
  amount        VARCHAR(32) NULL,
  unit          VARCHAR(16) NULL,
  category      VARCHAR(32) NOT NULL DEFAULT '其他' COMMENT '分组:调味/蔬菜/肉类/其他',
  checked       TINYINT NOT NULL DEFAULT 0 COMMENT '1已购',
  recipe_source VARCHAR(64) NULL COMMENT '来自哪个菜谱',
  sort          INT NOT NULL DEFAULT 0,
  PRIMARY KEY (id),
  KEY idx_sli_list (list_id)
) ENGINE=InnoDB COMMENT='购物清单项';

CREATE TABLE t_merchant (
  id               BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  name             VARCHAR(64) NOT NULL,
  logo             VARCHAR(255)    NULL,
  rating           INT     NOT NULL DEFAULT 95 COMMENT '好评率(%)',
  distance_km      DECIMAL(4,1) NOT NULL DEFAULT 1.0,
  delivery_minutes INT     NOT NULL DEFAULT 30 COMMENT '约送达(分钟)',
  tags             VARCHAR(64) NULL COMMENT '标签:综合推荐/价格最低',
  stock_status     VARCHAR(16) NOT NULL DEFAULT '库存充足' COMMENT '库存充足/库存紧张',
  audit_status     VARCHAR(16) NOT NULL DEFAULT '已通过' COMMENT '待审核/已通过/已驳回',
  status           TINYINT NOT NULL DEFAULT 1,
  create_time      DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  KEY idx_merchant_audit (audit_status)
) ENGINE=InnoDB COMMENT='商家';

CREATE TABLE t_merchant_product (
  id            BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  merchant_id   BIGINT UNSIGNED NOT NULL,
  ingredient_id BIGINT UNSIGNED    NULL,
  name          VARCHAR(64) NOT NULL,
  price         DECIMAL(8,2) NOT NULL DEFAULT 0,
  sell_unit     VARCHAR(32) NOT NULL DEFAULT '1 份' COMMENT '售卖规格（购买单位）',
  stock         INT NOT NULL DEFAULT 100,
  audit_status  VARCHAR(16) NOT NULL DEFAULT '已通过' COMMENT '商品上架审核',
  status        TINYINT NOT NULL DEFAULT 1,
  PRIMARY KEY (id),
  KEY idx_mp_merchant (merchant_id),
  KEY idx_mp_audit (audit_status)
) ENGINE=InnoDB COMMENT='商家商品报价';

CREATE TABLE t_orders (
  id            BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  order_no      VARCHAR(32) NOT NULL,
  user_id       BIGINT UNSIGNED NOT NULL,
  merchant_id   BIGINT UNSIGNED    NULL,
  merchant_name VARCHAR(64) NULL,
  amount        DECIMAL(8,2) NOT NULL DEFAULT 0,
  status        VARCHAR(16) NOT NULL DEFAULT '待支付' COMMENT '待支付/已支付/配送中/已完成',
  delivery_desc VARCHAR(128) NULL,
  create_time   DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  UNIQUE KEY uk_order_no (order_no),
  KEY idx_order_user (user_id),
  KEY idx_order_status (status)
) ENGINE=InnoDB COMMENT='订单';

CREATE TABLE t_order_item (
  id            BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  order_id      BIGINT UNSIGNED NOT NULL,
  ingredient_id BIGINT UNSIGNED    NULL,
  name          VARCHAR(64) NOT NULL,
  amount        VARCHAR(32) NULL,
  unit          VARCHAR(16) NULL,
  price         DECIMAL(8,2) NOT NULL DEFAULT 0,
  PRIMARY KEY (id),
  KEY idx_oi_order (order_id)
) ENGINE=InnoDB COMMENT='订单明细';

CREATE TABLE t_order_review (
  id          BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  order_id    BIGINT UNSIGNED NOT NULL,
  user_id     BIGINT UNSIGNED NOT NULL,
  rating      TINYINT NOT NULL DEFAULT 5,
  content     VARCHAR(500) NULL,
  create_time DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  KEY idx_or_order (order_id)
) ENGINE=InnoDB COMMENT='订单评价';

-- =====================================================================
-- 五、会员 / AI / 反馈 / 系统
-- =====================================================================
CREATE TABLE t_membership (
  id         BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  user_id    BIGINT UNSIGNED NOT NULL,
  plan_type  VARCHAR(16) NOT NULL DEFAULT '免费' COMMENT '免费/会员',
  start_time DATETIME     NULL,
  end_time   DATETIME     NULL,
  status     TINYINT NOT NULL DEFAULT 1,
  PRIMARY KEY (id),
  KEY idx_ms_user (user_id)
) ENGINE=InnoDB COMMENT='会员状态';

CREATE TABLE t_billing_record (
  id          BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  user_id     BIGINT UNSIGNED NOT NULL,
  type        VARCHAR(32) NOT NULL COMMENT '单次AI菜谱/会员·1个月',
  amount      DECIMAL(8,2) NOT NULL DEFAULT 0,
  pay_method  VARCHAR(16) NOT NULL DEFAULT '模拟支付',
  status      VARCHAR(16) NOT NULL DEFAULT '待支付' COMMENT '已支付/待支付',
  create_time DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  KEY idx_br_user (user_id)
) ENGINE=InnoDB COMMENT='订阅与购买记录';

CREATE TABLE t_ai_generation_record (
  id                BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  user_id           BIGINT UNSIGNED NOT NULL,
  source            VARCHAR(16) NOT NULL DEFAULT '会员AI新菜谱' COMMENT '首页生成/会员AI新菜谱',
  input_ingredients VARCHAR(255) NULL COMMENT '输入食材',
  recipe_id         BIGINT UNSIGNED NULL COMMENT '生成的菜谱',
  create_time       DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  KEY idx_ai_time (create_time)
) ENGINE=InnoDB COMMENT='AI 菜谱生成记录';

CREATE TABLE t_feedback (
  id          BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  user_id     BIGINT UNSIGNED    NULL,
  content     VARCHAR(500) NOT NULL,
  contact     VARCHAR(64)      NULL,
  status      VARCHAR(16) NOT NULL DEFAULT '待处理' COMMENT '待处理/已处理',
  create_time DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  KEY idx_fb_status (status)
) ENGINE=InnoDB COMMENT='意见反馈';

CREATE TABLE t_sys_config (
  id           BIGINT UNSIGNED NOT NULL AUTO_INCREMENT,
  config_key   VARCHAR(64)  NOT NULL,
  config_value VARCHAR(255) NOT NULL,
  remark       VARCHAR(128)     NULL,
  update_time  DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
  PRIMARY KEY (id),
  UNIQUE KEY uk_cfg_key (config_key)
) ENGINE=InnoDB COMMENT='系统配置';

-- =====================================================================
-- 补充商家在售商品（2026-10-27）
-- 背景：t_merchant_product 里「已通过」的只有 6 种调味料，
--       番茄/鸡蛋/豆腐/五花肉/胡萝卜 等生鲜全是「待审核」，
--       导致菜谱「一键下单」生成的订单里大部分食材价格为 0，看着像免费。
-- 本脚本为两家在营生鲜店补充已上架商品，保留原有待审核数据用于后台审核演示。
-- 可重复执行（按 merchant_id + name 去重）。
-- =====================================================================



-- 邻里鲜蔬铺（merchant 1）
INSERT INTO t_merchant_product (merchant_id, ingredient_id, name, price, sell_unit, stock, audit_status, status)
SELECT 1, i.id, t.name, t.price, t.unit, t.stock, '已通过', 1
FROM (
  SELECT '豆腐'   AS name, 3.80  AS price, '1 块(400g)' AS unit, 120 AS stock UNION ALL
  SELECT '胡萝卜', 2.60, '500g', 150 UNION ALL
  SELECT '五花肉', 18.00, '500g', 60  UNION ALL
  SELECT '生抽',   9.90, '1 瓶(500ml)', 80 UNION ALL
  SELECT '盐',     2.50, '1 袋(400g)', 200 UNION ALL
  SELECT '牛腩',   42.00, '500g', 30  UNION ALL
  SELECT '鸡胸肉', 15.00, '500g', 50  UNION ALL
  SELECT '大米',   25.00, '5kg',  70  UNION ALL
  SELECT '莲藕',   6.50, '500g', 40
) t
JOIN t_ingredient i ON i.name = t.name
WHERE NOT EXISTS (
  SELECT 1 FROM t_merchant_product p WHERE p.merchant_id = 1 AND p.name = t.name
);
